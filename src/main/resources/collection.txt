## Oficina API - cURL collection
## Base setup
# BASE_URL=http://localhost:8081
# TOKEN=COLE_SEU_JWT_AQUI
# CLIENTE_ID=00000000-0000-0000-0000-000000000001
# VEICULO_ID=00000000-0000-0000-0000-000000000002
# SERVICO_ID=00000000-0000-0000-0000-000000000003
# ITEM_ESTOQUE_ID=00000000-0000-0000-0000-000000000004
# ORDEM_ID=00000000-0000-0000-0000-000000000005

## AUTH
curl --request POST "$BASE_URL/auth/login" --header "Content-Type: application/json" --data "{\"login\":\"admin\",\"senha\":\"123456\"}"

## CLIENTES
curl --request POST "$BASE_URL/clientes" --header "Authorization: Bearer $TOKEN" --header "Content-Type: application/json" --data "{\"nome\":\"Maria Silva\",\"cpfCnpj\":\"12345678901\",\"email\":\"maria@email.com\",\"telefone\":\"11999999999\"}"
curl --request PUT "$BASE_URL/clientes/$CLIENTE_ID" --header "Authorization: Bearer $TOKEN" --header "Content-Type: application/json" --data "{\"nome\":\"Maria Atualizada\",\"email\":\"maria.atualizada@email.com\",\"telefone\":\"11888888888\"}"

curl --request GET "$BASE_URL/clientes/documento/12345678901" --header "Authorization: Bearer $TOKEN"
curl --request GET "$BASE_URL/clientes" --header "Authorization: Bearer $TOKEN"
curl --request DELETE "$BASE_URL/clientes/$CLIENTE_ID" --header "Authorization: Bearer $TOKEN"
curl --request GET "$BASE_URL/clientes/$CLIENTE_ID/veiculos" --header "Authorization: Bearer $TOKEN"
